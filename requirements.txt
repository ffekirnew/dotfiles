# Command-line tools needed by these dotfiles.
# Install these with your package manager (brew/apt/dnf/pacman), then run ./bootstrap.sh;
# bootstrap scripts do setup only and won't install these.
# Package names are Debian/Ubuntu unless noted.

# Core (used by more than one dotfiles directory)
zsh           # login shell; bin/*.sh use #!/usr/bin/env zsh
git           # vim.pack, TPM and oh-my-zsh clones; git/gitconfig
gh            # git credential helper (git/gitconfig); nvim gh-dash terminal
neovim        # >= 0.12 (vim.pack); apt often ships older, use brew/pacman/dnf or nvim/install.sh
ripgrep       # rg; telescope live_grep and grep_string
fd            # telescope find_files; Debian/Ubuntu: fd-find (binary: fdfind); Fedora: fd-find
python3       # >= 3.9 (bin/orp.py); mason pip tools need venv (Debian/Ubuntu: python3-venv)
make          # telescope-fzf-native and luasnip (make install_jsregexp) builds
gcc           # C compiler for treesitter parsers; Debian/Ubuntu: build-essential; Arch: base-devel
curl          # mason downloads; nvm install script
tar           # mason extracts archives
unzip         # mason extracts archives

# zsh (zsh/.zshrc)
fzf           # fzf --zsh key bindings and completion; >= 0.48
bat           # cat alias, fzf previews; Debian/Ubuntu: bat (binary: batcat)
eza           # ls alias, fzf cd preview; not in older apt repos, use release or cargo
dnsutils      # dig for fzf ssh preview; Fedora: bind-utils; Arch: bind; macOS: built in
nvm           # sourced from ~/.nvm/nvm.sh; not a package, use nvm's install script (curl)

# tmux (tmux/tmux.conf)
tmux          # prefix C-n; tmux-256color terminfo (ncurses)
xclip         # tmux-yank clipboard on X11; or wl-clipboard (wl-copy) on Wayland; macOS: pbcopy built in

# Neovim (nvim/)
# Mason installs basedpyright, ruff, lua-language-server and stylua itself
# (plugins/editor/lsp/mason.lua); only prerequisites above (basedpyright needs python3-venv).
lazygit       # <leader>lg terminal (plugins/terminals/toggleterm.lua); not in older apt repos
gh-dash       # gh extension: gh extension install dlvhdr/gh-dash (toggleterm.lua); needs gh
tree-sitter-cli # builds parsers for nvim-treesitter main (editor/treesitter/init.lua); brew/pacman: tree-sitter-cli

# Terminals (optional, only if you use them)
alacritty     # terminal emulator; alacritty/alacritty.toml
zellij        # multiplexer; zellij/config.kdl (minimum version not verified)

# Non-CLI (install manually)
# FiraCode Nerd Font Mono: alacritty.toml font; any Nerd Font works if you change the family

# Installed by ./bootstrap.sh (don't install these yourself):
# oh-my-zsh, powerlevel10k, zsh-autosuggestions (zsh/), TPM + tmux plugins (tmux/),
# alacritty-theme (alacritty/), Neovim plugins via vim.pack and LSP tools via Mason (nvim/)
